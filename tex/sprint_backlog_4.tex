% Epic 13: Calendrier et planification
\multicolumn{4}{|l|}{\cellcolor{blue!20}\textbf{Epic 13 - Calendrier et planification}} \\
\hline

US66 & En tant qu'\textbf{utilisateur}, je souhaite visualiser les tickets et demandes de maintenance sur un calendrier mensuel afin d'avoir une vue globale des interventions. &
\begin{itemize}[nosep, left=0pt, label={$\bullet$}]
\item S'appuyer sur les listes de tickets et de demandes déjà exposées par l'API et les afficher dans la vue calendrier de l'application.
\item Rendre les éléments dans la page \texttt{/calendrier} via les vues \texttt{month} et \texttt{agenda} avec code couleur par type d'élément.
\item Distinguer tickets et demandes directement dans le composant \texttt{GanttAgendaView}.
\end{itemize} & 3 \\
\hline

US67 & En tant qu'\textbf{administrateur}, je souhaite identifier visuellement les urgences SLA sur le calendrier afin de prioriser les interventions critiques. &
\begin{itemize}[nosep, left=0pt, label={$\bullet$}]
\item Calculer l'urgence à partir des dates d'échéance des tickets et des indicateurs SLA déjà présents dans le dashboard.
\item Mettre en évidence les tickets critiques ou expirés dans le calendrier sans créer de route dédiée supplémentaire.
\item Afficher une légende claire pour les statuts de priorité et de retard.
\end{itemize} & 2 \\
\hline

US68 & En tant que \textbf{chef de projet}, je souhaite accéder à une vue Gantt / Agenda afin d'analyser la charge de travail et la progression de chaque développeur. &
\begin{itemize}[nosep, left=0pt, label={$\bullet$}]
\item Exploiter \textbf{GET /api/tickets/developer-workload} et \textbf{GET /api/tickets/developer-schedule} pour alimenter la vue de planning.
\item Réutiliser le composant \texttt{GanttAgendaView} pour afficher la charge par développeur et la progression des tickets.
\item Permettre un filtrage temporel par jour, semaine et mois dans l'interface calendrier.
\end{itemize} & 3 \\
\hline

US69 & En tant qu'\textbf{utilisateur}, je souhaite accéder aux détails d'un ticket directement depuis le calendrier afin de consulter ou modifier l'intervention rapidement. &
\begin{itemize}[nosep, left=0pt, label={$\bullet$}]
\item Ouvrir les fiches ticket ou demande depuis le calendrier via \textbf{GET /api/tickets/:id} et \textbf{GET /api/demandes/:id}.
\item Utiliser \textbf{GET /api/tickets/:id/history} pour afficher la timeline d'un ticket depuis sa fiche détaillée.
\item Permettre un accès rapide à la consultation et à la mise à jour depuis la vue calendrier.
\end{itemize} & 2 \\
\hline

US70 & En tant qu'\textbf{utilisateur}, je souhaite accéder à une messagerie intégrée afin de communiquer avec mon équipe. &
\begin{itemize}[nosep, left=0pt, label={$\bullet$}]
\item S'appuyer sur \textbf{GET /api/channels/my}, \textbf{GET /api/channels/ticket/:ticketId} et le namespace WebSocket \texttt{messages}.
\item Afficher les canaux personnels, les canaux liés aux tickets et les messages directs.
\item Synchroniser l'interface avec le temps réel pour refléter les nouveaux messages et les présences.
\end{itemize} & 3 \\
\hline

US71 & En tant qu'\textbf{utilisateur}, je souhaite envoyer des messages dans un Canal afin de communiquer. &
\begin{itemize}[nosep, left=0pt, label={$\bullet$}]
\item Utiliser \textbf{POST /api/messages/channel/:channelId} pour envoyer un message dans un canal.
\item Gérer aussi l'envoi avec pièces jointes via \textbf{POST /api/messages/channel/:channelId/with-attachments}.
\item Diffuser les nouveaux messages en temps réel via l'événement \texttt{message:new}.
\end{itemize} & 2 \\
\hline

US72 & En tant qu'\textbf{utilisateur}, je souhaite répondre à un message afin de créer un fil de discussion. &
\begin{itemize}[nosep, left=0pt, label={$\bullet$}]
\item Réutiliser \textbf{POST /api/messages/channel/:channelId} avec \texttt{parentMessageId} pour créer une réponse liée.
\item Afficher le fil associé à partir de \textbf{GET /api/messages/:messageId/history}.
\item Conserver la lisibilité des échanges en chaîne dans la vue message.
\end{itemize} & 2 \\
\hline

US73 & En tant qu'\textbf{utilisateur}, je souhaite mentionner des collègues afin d'attirer leur attention. &
\begin{itemize}[nosep, left=0pt, label={$\bullet$}]
\item Détecter les mentions lors de l'envoi d'un message via \textbf{POST /api/messages/channel/:channelId}.
\item Centraliser les mentions reçues avec \textbf{GET /api/messages/mentions}.
\item Permettre le marquage comme lue via \textbf{PATCH /api/messages/mentions/:mentionId/read}.
\end{itemize} & 2 \\
\hline

US74 & En tant qu'\textbf{utilisateur}, je souhaite éditer mes messages afin de corriger des erreurs. &
\begin{itemize}[nosep, left=0pt, label={$\bullet$}]
\item Autoriser la modification d'un message avec \textbf{PATCH /api/messages/:messageId}.
\item Rafraîchir le fil en temps réel via l'événement \texttt{message:updated}.
\item Conserver l'historique d'édition côté service pour la traçabilité.
\end{itemize} & 1 \\
\hline

US75 & En tant qu'\textbf{utilisateur}, je souhaite supprimer mes messages afin de retirer du contenu inapproprié. &
\begin{itemize}[nosep, left=0pt, label={$\bullet$}]
\item Supprimer un message via \textbf{DELETE /api/messages/:messageId}.
\item Diffuser la suppression en temps réel avec \texttt{message:deleted}.
\item Masquer le contenu supprimé tout en conservant la cohérence du fil.
\end{itemize} & 1 \\
\hline

US76 & En tant qu'\textbf{utilisateur}, je souhaite épingler des messages importants afin de les retrouver facilement. &
\begin{itemize}[nosep, left=0pt, label={$\bullet$}]
\item Épingler ou retirer l'épinglage via \textbf{POST /api/messages/:messageId/pin}.
\item Consulter les messages épinglés avec \textbf{GET /api/messages/channel/:channelId/pinned}.
\item Mettre ces messages en avant dans le canal concerné.
\end{itemize} & 1 \\
\hline

US77 & En tant que \textbf{chef de projet}, je souhaite voir qui est en ligne dans mon équipe afin de savoir qui je peux solliciter. &
\begin{itemize}[nosep, left=0pt, label={$\bullet$}]
\item Exploiter la charge et la disponibilité de l'équipe via \textbf{GET /api/dashboard/chef-projet/equipe-charge}.
\item Afficher les membres en travail ou disponibles directement dans la vue chef de projet.
\item Utiliser les états temps réel pour compléter l'information de disponibilité.
\end{itemize} & 2 \\
\hline

US78 & En tant qu'\textbf{utilisateur}, je souhaite voir qui est en ligne afin de savoir qui est disponible. &
\begin{itemize}[nosep, left=0pt, label={$\bullet$}]
\item Diffuser le statut via le namespace WebSocket \texttt{messages} et l'événement \texttt{user:status}.
\item Mettre à jour la dernière connexion côté backend à la connexion WebSocket.
\item Afficher un état simple en ligne / hors ligne dans l'interface de messagerie.
\end{itemize} & 2 \\
\hline

US79 & En tant qu'\textbf{utilisateur}, je souhaite uploader des documents afin de centraliser les fichiers. &
\begin{itemize}[nosep, left=0pt, label={$\bullet$}]
\item Utiliser \textbf{POST /api/documents/upload} pour déposer un ou plusieurs fichiers.
\item Associer les documents à un ticket ou à un contrat via \texttt{ticketId} ou \texttt{contratId}.
\item Respecter les types et tailles de fichiers autorisés par l'API.
\end{itemize} & 3 \\
\hline

US80 & En tant qu'\textbf{utilisateur}, je souhaite télécharger des documents afin de les consulter hors ligne. &
\begin{itemize}[nosep, left=0pt, label={$\bullet$}]
\item S'appuyer sur la liste des documents et sur l'URL du fichier renvoyée par l'objet document.
\item Enregistrer la consultation avec \textbf{POST /api/documents/:id/consulter}.
\item Ne pas inventer de route de téléchargement dédiée, l'accès au fichier se fait via l'URL exposée.
\end{itemize} & 2 \\
\hline

US81 & En tant qu'\textbf{utilisateur}, je souhaite gérer les versions des documents afin d'assurer la traçabilité. &
\begin{itemize}[nosep, left=0pt, label={$\bullet$}]
\item Consulter l'historique avec \textbf{GET /api/documents/:id/versions}.
\item Remplacer un document par une nouvelle version avec \textbf{PATCH /api/documents/:id/replace}.
\item Conserver l'ancienne version pour assurer la traçabilité.
\end{itemize} & 2 \\
\hline

US82 & En tant qu'\textbf{utilisateur}, je souhaite consulter l'historique des consultations afin de voir qui a accédé au document. &
\begin{itemize}[nosep, left=0pt, label={$\bullet$}]
\item Enregistrer et lister les consultations via \textbf{POST /api/documents/:id/consulter} et \textbf{GET /api/documents/:id/consultations}.
\item Afficher les statistiques de consultation avec \textbf{GET /api/documents/:id/consultations/stats}.
\item Restituer une frise chronologique avec les utilisateurs et les dates d'accès.
\end{itemize} & 1 \\
\hline

US83 & En tant qu'\textbf{administrateur}, je souhaite archiver des documents afin de nettoyer la vue active. &
\begin{itemize}[nosep, left=0pt, label={$\bullet$}]
\item Archiver un document avec \textbf{PATCH /api/documents/:id/archive}.
\item Restaurer un document avec \textbf{PATCH /api/documents/:id/restore}.
\item Réserver ces actions aux rôles administrateur.
\end{itemize} & 1 \\
\hline

US84 & En tant que \textbf{chef de projet}, je souhaite générer automatiquement un PV à partir d'un ticket afin de documenter les interventions. &
\begin{itemize}[nosep, left=0pt, label={$\bullet$}]
\item Préremplir les données du PV avec \textbf{GET /api/pv/prefill/:ticketId}.
\item Générer le PV avec \textbf{POST /api/pv/generate/:ticketId}.
\item Lier le PV au ticket source et conserver les données utiles à la rédaction.
\end{itemize} & 3 \\
\hline

US85 & En tant que \textbf{client}, je souhaite télécharger les PV de mes tickets afin de conserver une trace des interventions. &
\begin{itemize}[nosep, left=0pt, label={$\bullet$}]
\item Générer le PV depuis le ticket via \textbf{POST /api/pv/generate/:ticketId}.
\item Exposer le document final dans l'interface client ou ticket après génération.
\item Conserver la trace du PV dans le workflow du ticket.
\end{itemize} & 1 \\
\hline

US86 & En tant que \textbf{développeur ou chef de projet}, je souhaite consulter la base de connaissances afin de trouver des solutions existantes. &
\begin{itemize}[nosep, left=0pt, label={$\bullet$}]
\item Consulter les articles via \textbf{GET /api/knowledge-base} et \textbf{GET /api/knowledge-base/:id}.
\item Rechercher des articles liés à un ticket avec \textbf{GET /api/knowledge-base/by-ticket/:ticketId}.
\item Utiliser les suggestions contextuelles via \textbf{POST /api/knowledge-base/suggest}.
\end{itemize} & 2 \\
\hline

US87 & En tant qu'\textbf{administrateur}, je souhaite créer des articles de la base de connaissances afin d'enrichir la documentation. &
\begin{itemize}[nosep, left=0pt, label={$\bullet$}]
\item Créer un article manuel avec \textbf{POST /api/knowledge-base}.
\item Mettre à jour ou archiver un article avec \textbf{PATCH /api/knowledge-base/:id} et \textbf{DELETE /api/knowledge-base/:id}.
\item Enregistrer l'usage des articles via \textbf{POST /api/knowledge-base/:id/usage}.
\end{itemize} & 2 \\
\hline

US88 & En tant qu'\textbf{utilisateur}, je souhaite créer une nouvelle version d'un document afin de maintenir l'historique. &
\begin{itemize}[nosep, left=0pt, label={$\bullet$}]
\item Créer une nouvelle version en remplaçant le fichier avec \textbf{PATCH /api/documents/:id/replace}.
\item Conserver les versions précédentes via \textbf{GET /api/documents/:id/versions}.
\item Assurer la continuité de l'historique documentaire.
\end{itemize} & 2 \\
\hline

US89 & En tant qu'\textbf{utilisateur}, je souhaite voir qui a consulté un document afin de suivre sa diffusion. &
\begin{itemize}[nosep, left=0pt, label={$\bullet$}]
\item Exploiter \textbf{GET /api/documents/:id/consultations} pour afficher les accès détaillés.
\item Utiliser \textbf{GET /api/documents/:id/consultations/stats} pour voir les vues uniques et les dates.
\item Présenter les consults dans une timeline lisible.
\end{itemize} & 2 \\
\hline

US90 & En tant qu'\textbf{utilisateur}, je souhaite recevoir des notifications in-app afin d'être informé en temps réel. &
\begin{itemize}[nosep, left=0pt, label={$\bullet$}]
\item Lister les notifications via \textbf{GET /api/notifications}.
\item Marquer une notification comme lue avec \textbf{PATCH /api/notifications/:id/read} ou tout lire avec \textbf{PATCH /api/notifications/read-all}.
\item Mettre à jour le badge temps réel via le gateway de notifications.
\end{itemize} & 2 \\
\hline

US91 & En tant qu'\textbf{utilisateur}, je souhaite recevoir des notifications par email afin d'être informé hors connexion. &
\begin{itemize}[nosep, left=0pt, label={$\bullet$}]
\item Gérer les préférences email avec \textbf{GET /api/notifications/preferences} et \textbf{PATCH /api/notifications/preferences/:type}.
\item Déclencher les envois depuis les événements métier déjà existants.
\item Permettre l'activation ou la désactivation par type de notification.
\end{itemize} & 2 \\
\hline

US92 & En tant que \textbf{développeur ou chef de projet}, je souhaite recevoir des notifications Slack afin d'être alerté sur mon outil de travail. &
\begin{itemize}[nosep, left=0pt, label={$\bullet$}]
\item Configurer les intégrations Slack avec \textbf{POST /api/slack/config} et \textbf{GET /api/slack/configs}.
\item Consulter, modifier ou supprimer une configuration par contrat via \textbf{GET /api/slack/config/:contratId}, \textbf{PATCH /api/slack/config/:contratId} et \textbf{DELETE /api/slack/config/:contratId}.
\item Tester une configuration avec \textbf{POST /api/slack/config/:contratId/test}.
\end{itemize} & 2 \\
\hline

US93 & En tant qu'\textbf{utilisateur}, je souhaite recevoir des notifications push navigateur afin d'être alerté même hors onglet. &
\begin{itemize}[nosep, left=0pt, label={$\bullet$}]
\item Récupérer la clé publique avec \textbf{GET /api/web-push/vapid-public-key}.
\item S'abonner ou se désabonner via \textbf{POST /api/web-push/subscribe} et \textbf{DELETE /api/web-push/unsubscribe}.
\item Associer les abonnements aux événements métier et aux alertes temps réel.
\end{itemize} & 2 \\
\hline

US94 & En tant qu'\textbf{utilisateur}, je souhaite consulter l'historique de mes notifications afin de retrouver une information. &
\begin{itemize}[nosep, left=0pt, label={$\bullet$}]
\item S'appuyer sur la liste paginée \textbf{GET /api/notifications} comme historique utilisateur.
\item Exploiter les filtres de pagination et le compteur de non lus.
\item Présenter les notifications dans une vue chronologique simple.
\end{itemize} & 1 \\
\hline

US95 & En tant qu'\textbf{administrateur}, je souhaite consulter un dashboard global afin de piloter le système. &
\begin{itemize}[nosep, left=0pt, label={$\bullet$}]
\item Consulter les statistiques globales via \textbf{GET /api/dashboard/stats}.
\item Compléter le tableau de bord avec \textbf{GET /api/dashboard/top-developers}, \textbf{GET /api/dashboard/trend} et \textbf{GET /api/dashboard/rapport-mensuel}.
\item Afficher les alertes, les tickets critiques et les tendances de l'application.
\end{itemize} & 3 \\
\hline

US96 & En tant que \textbf{chef de projet}, je souhaite consulter un dashboard d'équipe afin de piloter l'activité. &
\begin{itemize}[nosep, left=0pt, label={$\bullet$}]
\item Utiliser \textbf{GET /api/dashboard/chef-projet/kpis} pour les indicateurs clés.
\item Compléter la vue avec \textbf{GET /api/dashboard/chef-projet/demandes-action-requise}, \textbf{GET /api/dashboard/chef-projet/tickets-sla-critique}, \textbf{GET /api/dashboard/chef-projet/projets-synthese} et \textbf{GET /api/dashboard/chef-projet/equipe-charge}.
\item Suivre l'évolution via \textbf{GET /api/dashboard/chef-projet/evolution} et \textbf{GET /api/dashboard/chef-projet/activite-recente}.
\end{itemize} & 2 \\
\hline

US97 & En tant qu'\textbf{client}, je souhaite consulter mon dashboard afin de suivre mes projets. &
\begin{itemize}[nosep, left=0pt, label={$\bullet$}]
\item Consulter le dashboard client via \textbf{GET /api/dashboard/client}.
\item Afficher les projets, tickets et indicateurs liés au client connecté.
\item Présenter une vue synthétique adaptée au compte client.
\end{itemize} & 2 \\
\hline

US98 & En tant qu'\textbf{administrateur}, je souhaite générer un rapport d'activité mensuel afin de présenter les statistiques. &
\begin{itemize}[nosep, left=0pt, label={$\bullet$}]
\item Générer le rapport via \textbf{GET /api/dashboard/rapport-mensuel} avec \texttt{mois} et \texttt{annee}.
\item Permettre un filtrage optionnel par client avec \texttt{clientId}.
\item Réutiliser les métriques déjà calculées par le dashboard admin.
\end{itemize} & 3 \\
\hline

US99 & En tant qu'\textbf{administrateur}, je souhaite générer des rapports de conformité SLA afin de justifier les performances. &
\begin{itemize}[nosep, left=0pt, label={$\bullet$}]
\item Combiner \textbf{GET /api/dashboard/stats} et \textbf{GET /api/dashboard/trend} pour calculer le taux de conformité SLA.
\item S'appuyer sur les tickets critiques, les retards et les blocages déjà agrégés par le dashboard.
\item Ne pas prévoir de route SLA-report séparée, car la conformité est calculée à partir des vues existantes.
\end{itemize} & 2 \\
\hline

US100 & En tant que \textbf{chef de projet}, je souhaite générer des rapports d'activité afin de présenter l'avancement. &
\begin{itemize}[nosep, left=0pt, label={$\bullet$}]
\item Utiliser \textbf{GET /api/dashboard/chef-projet/rapport} avec \texttt{dateDebut}, \texttt{dateFin}, \texttt{projetId} et \texttt{clientId}.
\item Produire un rapport par projet ou par période à partir des données réelles de tickets et de demandes.
\item Réutiliser les métriques d'évolution et d'activité déjà exposées au chef de projet.
\end{itemize} & 2 \\
