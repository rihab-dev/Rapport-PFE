% Epic 1: Authentification et Sécurité
\multicolumn{4}{|l|}{\cellcolor{blue!20}\textbf{Epic 1 - Authentification et Sécurité}} \\
\hline

US01 & En tant qu'\textbf{utilisateur}, je souhaite m'authentifier avec email et mot de passe afin d'accéder à mes informations de manière sécurisée. & 
\vspace{2mm}
\begin{itemize}[nosep, left=0pt, label={$\bullet$}]
\item Développer l'endpoint "POST /api/auth/login"
\item Mettre en place la gestion de session via \textbf{JWT} (access token 15 min, refresh token 7 jours stocké en base)
\item Réaliser l'interface connexion login
\item Consommer l'API d'authentification
\item Gérer les tentatives échouées (3 essais) avec verrouillage temporaire
\end{itemize} & 2 \\
\hline

US02 & En tant qu'\textbf{utilisateur}, je souhaite réinitialiser mon mot de passe afin de retrouver l'accès à mon compte. & 
\vspace{2mm}
\begin{itemize}[nosep, left=0pt, label={$\bullet$}]
\item Développer les endpoints "POST /api/auth/mot-de-passe-oublie et "POST /api/auth/reinitialiser-mot-de-passe"
  \item Générer un token sécurisé (bcrypt) avec expiration de \textbf{1 heure}
  \item Invalider les anciens tokens non utilisés à chaque nouvelle demande
  \item Configurer l'envoi d'un e-mail de réinitialisation et d'un e-mail de confirmation via \textbf{Nodemailer}
  \item Créer les formulaires « mot de passe oublié » et de réinitialisation côté Next.js

\end{itemize} & 2 \\
\hline

US03 & En tant qu'\textbf{utilisateur}, je souhaite m'authentifier via Google ou GitHub afin de me connecter rapidement sans créer de compte. & 
\vspace{2mm}
\begin{itemize}[nosep, left=0pt, label={$\bullet$}]
\item Intégrer les stratégies OAuth2 via "passport-google-oauth20" et "passport-github2" dans NestJS (\textbf{@nestjs/passport})
\item Implémenter les routes "GET /api/auth/google", "GET /api/auth/github" et leurs callbacks
  \item Si l'utilisateur existe en base et est actif : générer les tokens JWT et rediriger vers le frontend avec \textbf{?oauth=success}
\item Vérifier l'existence de l'utilisateur en base de données et une demande d'inscription OAuth est envoyée à l'admin
\item Implémenter les boutons OAuth sur la page de login
\end{itemize} & 3 \\
\hline

US04 & En tant qu'\textbf{utilisateur}, je souhaite me déconnecter de la plateforme afin de sécuriser ma session. & 
\vspace{2mm}
\begin{itemize}[nosep, left=0pt, label={$\bullet$}]
\item Développer l'endpoint "POST /auth/logout" avec révocation du refresh token en base
\item Supprimer \textbf{accessToken}, \textbf{refreshToken} et \textbf{user} du \textbf{localStorage}
\item Vider le cache React Query (\textbf{queryClient.clear()})
\item Réaliser le bouton de déconnexion 
\end{itemize} & 1 \\
\hline

US05 & En tant qu'\textbf{utilisateur}, je souhaite que mon compte soit protégé contre les tentatives de connexion malveillantes afin de garantir la sécurité de mes données. & 
\vspace{2mm}
\begin{itemize}[nosep, left=0pt, label={$\bullet$}]
  \item Implémenter le compteur de tentatives échouées en base de données
  \item Verrouiller le compte après pendant \textbf{1 minute} 
  \item Réinitialiser le compteur à 0 après une connexion réussie
  \item Appliquer le \textbf{rate limiting} global via \textbf{@nestjs/throttler} sur les endpoints publics
\end{itemize} & 2 \\
\hline

% Epic 2: Gestion des utilisateurs
\multicolumn{4}{|l|}{\cellcolor{blue!20}\textbf{Epic 2 - Gestion des utilisateurs}} \\
\hline

US06 & En tant qu'\textbf{utilisateur}, je souhaite gérer mon profil afin que mes informations soient toujours à jour. & 
\vspace{2mm}
\begin{itemize}[nosep, left=0pt, label={$\bullet$}]
\item Développer les endpoints "GET /api/profil" et "PUT /api/profil" pour consulter et modifier les informations
\item Implémenter l'upload de la photo de profil "POST /api/profil/upload-photo"
\item Développer le changement de mot de passe via "POST /api/profil/change-password" avec vérification de l'ancien mot de passe (bcrypt)
\item Gérer les e-mails secondaires : "GET/POST/DELETE /api/profil/emails" et "PATCH /api/profil/emails/principal"
\item Réaliser l'interface profil côté Next.js  avec formulaire prérempli
\end{itemize} & 2 \\
\hline

US07 & En tant qu'\textbf{utilisateur}, je souhaite consulter l'historique de mes activités afin de suivre mes actions dans le système. & 
\vspace{2mm}
\begin{itemize}[nosep, left=0pt, label={$\bullet$}]
  \item Développer l'endpoint
    "GET /api/activity-logs/me"
    retournant les activités de l'utilisateur connecté
    (limité aux \textbf{30 derniers jours} par défaut)
  \item Développer "GET /api/activity-logs/me/stats"
    pour les statistiques personnelles (taux de succès,
    actions fréquentes, activité par jour)
  \item Développer "GET /api/activity-logs/me/export"
    pour l'export CSV ou JSON
  \item Implémenter le filtrage par \textbf{entityType},
    \textbf{action}, \textbf{status}, plage de dates
    et recherche textuelle (\textbf{contains})
  
  \item Réaliser l'interface timeline avec filtres
    côté Next.js
\end{itemize} & 2 \\
\hline
US08 & En tant qu'\textbf{administrateur}, je souhaite créer des utilisateurs afin de gérer les accès au système. & 
\vspace{2mm}
\begin{itemize}[nosep, left=0pt, label={$\bullet$}]
\item Développer l'endpoint "POST /api/users"
\item Générer automatiquement un mot de passe temporaire haché via \textbf{bcrypt}
\item Générer un token de réinitialisation (1 heure) et construire le lien de réinitialisation
\item Créer automatiquement le "ProfilEmploye" avec matricule "EMP-\{id\}" 
\item Valider les données d'entrée via \textbf{class-validator}
\item Envoyer un e-mail de bienvenue via \textbf{Nodemailer} contenant le mot de passe temporaire et le lien de réinitialisation
\item Réaliser le formulaire d'ajout côté Next.js
\end{itemize} & 3 \\
\hline

US09 & En tant qu'\textbf{administrateur}, je souhaite modifier les informations d'un utilisateur afin de maintenir des données à jour. & 
\vspace{2mm}
\begin{itemize}[nosep, left=0pt, label={$\bullet$}]
\item Développer l'endpoint "PUT /api/users/:id"
\item Valider les données via \textbf{class-validator}
\item Réaliser le formulaire de modification côté Next.js
\end{itemize} & 2 \\
\hline

US10 & En tant qu'\textbf{administrateur}, je souhaite activer et désactiver un compte utilisateur afin de contrôler l'accès au système. & 
\vspace{2mm}
\begin{itemize}[nosep, left=0pt, label={$\bullet$}]
\item Développer les endpoints "PATCH /api/users/:id/desactiver" et "PATCH /api/users/:id/reactiver"
\item Implémenter les vérifications de réassignation des tickets et des demandes en cas de désactivation
\item Envoyer des notifications pour informer l'utilisateur de l'activation ou désactivation de son compte
\item Réaliser les boutons d'action avec appels des API
\end{itemize} & 3 \\
\hline

US11 & En tant qu'\textbf{administrateur}, je souhaite consulter la liste des utilisateurs afin de visualiser et gérer efficacement les comptes. & 
\vspace{2mm}
\begin{itemize}[nosep, left=0pt, label={$\bullet$}]
  \item Développer l'endpoint "GET /api/users"
     retournant les
    utilisateurs hors rôles CLIENT et ADMIN, hors
    utilisateur connecté
  \item Développer l'endpoint de recherche
    "GET /api/users/search?q=\&limit="
    avec filtre insensible à la casse
  \item Réaliser l'interface d'administration avec vue grille
    et vue liste, filtres par statut et par rôle,
    barre de recherche dynamique
\end{itemize} & 1 \\
\hline



% Epic 3: Gestion des rôles et permissions
\multicolumn{4}{|l|}{\cellcolor{blue!20}\textbf{Epic 3 - Gestion des rôles et permissions}} \\
\hline

US12 & En tant qu'\textbf{administrateur}, je souhaite gérer des rôles afin de structurer les accès au système. & 
\vspace{2mm}
\begin{itemize}[nosep, left=0pt, label={$\bullet$}]
  \item Développer les endpoints CRUD sous
    "/api/admin/roles-permissions/roles" :
    \textbf{POST}, \textbf{GET}, \textbf{PATCH /:id},
    \textbf{DELETE /:id}
  \item Implémenter l'archivage "PATCH /:id/archive"
    et la restauration "PATCH /:id/restore"
  \item Attribuer automatiquement \textbf{ permissions de base}
    à chaque nouveau rôle 
  \item Protéger les rôles système (\textbf{roleSysteme = true})
    contre la suppression et le renommage
  \item Implémenter le \textbf{PermissionsRolesGuard} et le
    décorateur \textbf{@Permissions()} pour sécuriser
    les routes
  \item Réaliser l'interface de gestion des rôles côté Next.js
\end{itemize} & 3 \\
\hline

US13 & En tant qu'\textbf{administrateur}, je souhaite attribuer des permissions granulaires aux rôles afin de contrôler précisément les accès. & 
\vspace{2mm}
\begin{itemize}[nosep, left=0pt, label={$\bullet$}]
  \item Développer les endpoints d'attribution et de retrait :
    "POST /api/admin/roles-permissions/roles/:roleId/permissions"
    et
    "DELETE /api/admin/roles-permissions/roles/:roleId/permissions/:permissionId"
  \item Développer l'endpoint d'attribution directe :
    "POST /api/admin/roles-permissions/assign-permission"
  \item Implémenter la détection des doublons
    
  \item Développer l'interface interactive de type
    « Matrice de contrôle d'accès » avec cases à cocher
    organisées par catégorie
\end{itemize} & 2 \\
\hline
US14 & En tant qu'\textbf{administrateur}, je souhaite consulter la matrice rôles-permissions afin de visualiser rapidement les droits de chaque rôle. & 
\vspace{2mm}
\begin{itemize}[nosep, left=0pt, label={$\bullet$}]
  \item Développer l'endpoint
    "GET /api/admin/roles-permissions/roles"
    retournant les rôles avec leurs permissions imbriquées
    
  \item Développer l'endpoint
    "GET /api/admin/roles-permissions/permissions"
    retournant toutes les permissions triées par
    \textbf{codePermission}
  \item Structurer les données côté frontend en matrice
    rôles × permissions avec cases cochées/décochées
  \item Réaliser l'interface de visualisation avec vue
    « Par rôle » et export de la matrice
\end{itemize} & 3 \\
\hline

% Epic 4: Demandes d'inscription OAuth
\multicolumn{4}{|l|}{\cellcolor{blue!20}\textbf{Epic 4 - Demandes d'inscription OAuth}} \\
\hline

US15 & En tant qu'\textbf{utilisateur externe}, je souhaite demander l'accès via OAuth afin de m'inscrire facilement. & 
\vspace{2mm}
\begin{itemize}[nosep, left=0pt, label={$\bullet$}]
  \item La demande est créée automatiquement dans la stratégie
    Passport (\textbf{GoogleStrategy} /
    \textbf{GithubStrategy}) lors du callback OAuth
  \item Enregistrer la demande en base
     avec statut
    \textbf{EN\_ATTENTE} et le rôle souhaité
    
  \item Détecter les doublons (même email + fournisseur)
    via \textbf{upsert} pour éviter les entrées multiples
  \item Notifier tous les administrateurs actifs par e-mail
  \item Envoyer un e-mail de confirmation au visiteur
  \item Afficher un message d'attente côté frontend
    
\end{itemize} & 3 \\
\hline


US16 & En tant qu'\textbf{administrateur}, je souhaite valider ou rejeter les demandes d'inscription OAuth afin de contrôler les accès. &
\vspace{2mm}
\begin{itemize}[nosep, left=0pt, label={$\bullet$}]
  \item Développer l'endpoint de liste :
    "GET /api/admin/demandes-inscription"
    avec filtre par statut 
    
  \item Développer l'endpoint d'approbation :
    "PATCH /api/admin/demandes-inscription/:id/approuver"
    — crée le compte utilisateur, attribue le rôle,
    marque la demande \textbf{APPROUVEE}
    
  \item Développer l'endpoint de refus :
    "PATCH /api/admin/demandes-inscription/:id/refuser"
    avec motif optionnel, marque la demande
    \textbf{REFUSEE}
    
  \item Configurer l'envoi d'e-mails de confirmation
    ou de refus via \textbf{Nodemailer}
  \item Réaliser l'interface de gestion des demandes
    avec actions d'approbation/refus côté Next.js
\end{itemize} & 3 \\
\hline


\end{longtable}
